FlowEngine — Engine Loop

Implementation reference for the core engine loop, background services, and DI registration.

Overview

The core engine loop — the orchestrator that takes an ExecutionInstance from one stable state to the next. Central piece that all node executors, state store implementations, and background services depend on.

StartAsync implementation

csharppublic async Task<ExecutionInstance> StartAsync(StartRequest request, CancellationToken ct = default)
{
    var workflowId       = request.WorkflowId;
    var entity           = request.Entity;
    var @params          = request.Params;
    var correlationId    = request.CorrelationId;
    var tags             = request.Tags;
    var deadline         = request.Deadline;
    var parentInstanceId = request.ParentInstanceId;
    var parentNodeId     = request.ParentNodeId;

    // 1. Load and validate workflow definition
    var definition = await _workflowRegistry.GetAsync(workflowId, version: null, ct)
        ?? throw new WorkflowNotFoundException(workflowId);

    if (definition.Status != DefinitionStatus.Active)
        throw new WorkflowNotActiveException(workflowId, definition.Status);

    // 2. Validate all client dependencies are registered and reachable
    await ValidateWorkflowDependenciesAsync(definition, ct);

    // 3. Validate params against definition.ParamsSchema if set
    if (!string.IsNullOrWhiteSpace(definition.ParamsSchema))
    {
        var paramsErrors = ParamsSchemaValidator.Validate(
            definition.ParamsSchema,
            @params as IReadOnlyDictionary<string, ContextValue>
                ?? @params?.ToDictionary(kv => kv.Key, kv => kv.Value));
        if (paramsErrors.Count > 0)
            throw new ArgumentException(
                $"Workflow '{workflowId}' params schema validation failed: "
                + string.Join("; ", paramsErrors));
    }

    // 4. Create instance
    var instance = new ExecutionInstance
    {
        InstanceId       = Guid.NewGuid().ToString("N"),
        WorkflowId       = definition.Id,
        WorkflowVersion  = definition.Version,
        CorrelationId    = correlationId,
        Status           = ExecStatus.Running,
        CurrentNodeId    = definition.EntryNodeId,
        CreatedAt        = DateTimeOffset.UtcNow,
        UpdatedAt        = DateTimeOffset.UtcNow,
        Deadline         = deadline,
        ParentInstanceId = parentInstanceId,
        ParentNodeId     = parentNodeId,
        Tags             = MergeTags(definition.DefaultTags, tags),
        Context          = new ContextDictionary
        {
            Entity  = entity,
            Params  = @params,
            Context = new Dictionary<string, ContextValue>()
        }
    };

    // 5. Save initial state (no claim needed — nobody else knows this ID yet)
    await _stateStore.SaveAsync(instance, ct);

    // 6. Emit telemetry
    EmitEvent("flow.instance.started", instance);

    // 7. Run the engine loop
    return await ExecuteLoopAsync(instance, ct);
}

// ── ValidateWorkflowDependenciesAsync ──────────────────────────────────────────

/// <summary>
/// Called before any instance starts. Scans all nodes for clientRef values,
/// verifies each is registered in the IClientRegistry, and calls ValidateAsync
/// on each to confirm connectivity.
///
/// Fails fast with a MissingClientException before any execution begins.
/// This prevents workflows with misconfigured dependencies from silently
/// failing on the first node that tries to use an unregistered client.
/// </summary>
private async Task ValidateWorkflowDependenciesAsync(
    WorkflowDefinition definition, CancellationToken ct)
{
    // Extract all clientRefs from node configs
    var clientRefs = definition.Nodes.Values
        .Select(n => ExtractClientRef(n))
        .Where(r => r != null)
        .Distinct()
        .ToList();

    foreach (var clientRef in clientRefs)
    {
        var client = _clientRegistry.Get(clientRef!)
            ?? throw new MissingClientException(
                $"Workflow '{definition.Id}' node requires client '{clientRef}' " +
                $"which is not registered. Register it via " +
                $"FlowEngineBuilder.AddClient(\"{clientRef}\", ...).");

        try
        {
            await client.ValidateAsync(ct);
        }
        catch (Exception ex)
        {
            throw new ClientValidationException(
                $"Client '{clientRef}' required by workflow '{definition.Id}' " +
                $"failed validation: {ex.Message}", ex);
        }
    }
}

/// <summary>
/// Extract the clientRef from a node's Config JsonElement.
/// Returns null for node types that don't use external clients.
/// </summary>
private static string? ExtractClientRef(NodeDefinition node)
{
    try
    {
        if (node.Config.ValueKind == JsonValueKind.Object
            && node.Config.TryGetProperty("clientRef", out var prop))
            return prop.GetString();
    }
    catch { /* ignore deserialization errors — caught at execution time */ }
    return null;
}

ResumeAsync implementation

csharppublic async Task<ExecutionInstance> ResumeAsync(
    string instanceId, string eventName, string correlationKey,
    JsonElement? signalPayload = null, CancellationToken ct = default)
{
    var instance = await _stateStore.FindBySuspendedEventAsync(eventName, correlationKey, ct);

    // Fallback: event lookup may miss if PendingWait was already cleared.
    // Return idempotently if the instance is already past suspension.
    if (instance == null)
    {
        var direct = await _stateStore.LoadAsync(instanceId, ct)
            ?? throw new InstanceNotFoundException(instanceId);
        if (direct.Status != ExecStatus.Suspended || direct.PendingWait == null)
            return direct;
        throw new InstanceNotFoundException(instanceId);
    }

    if (instance.Status != ExecStatus.Suspended)
        return instance; // idempotent

    if (instance.PendingWait == null)
        return instance; // corrupted state: suspended without pending wait

    // Append signal to ReceivedSignals
    instance.PendingWait.ReceivedSignals.Add(
        signalPayload ?? JsonSerializer.SerializeToElement(new {}));

    // Not enough signals yet — save and return
    if (instance.PendingWait.ReceivedSignals.Count < instance.PendingWait.MinSignalCount)
    {
        instance.UpdatedAt = DateTimeOffset.UtcNow;
        await _stateStore.SaveAsync(instance, ct);
        return instance;
    }

    // Enough signals — write payload to context, advance past the wait node, run loop
    AppendSignalToContext(instance, signalPayload);
    await AdvancePastCurrentNodeAsync(instance, DecisionOutcome.Match, ct);
    await _stateStore.SaveAsync(instance, ct);

    EmitEvent("flow.instance.resumed", instance);
    return await ExecuteLoopAsync(instance, ct);
}

// ── ResumeByInstanceIdAsync ────────────────────────────────────────────────────

public async Task<ExecutionInstance> ResumeByInstanceIdAsync(
    string instanceId, JsonElement? signalPayload = null, CancellationToken ct = default)
{
    var instance = await _stateStore.LoadAsync(instanceId, ct)
        ?? throw new InstanceNotFoundException(instanceId);

    if (instance.Status != ExecStatus.Suspended)
        return instance; // idempotent

    if (instance.PendingWait != null)
    {
        instance.PendingWait.ReceivedSignals.Add(
            signalPayload ?? JsonSerializer.SerializeToElement(new {}));

        if (instance.PendingWait.ReceivedSignals.Count < instance.PendingWait.MinSignalCount)
        {
            instance.UpdatedAt = DateTimeOffset.UtcNow;
            await _stateStore.SaveAsync(instance, ct);
            return instance;
        }
    }

    AppendSignalToContext(instance, signalPayload);
    await AdvancePastCurrentNodeAsync(instance, DecisionOutcome.Match, ct);
    await _stateStore.SaveAsync(instance, ct);

    EmitEvent("flow.instance.resumed", instance);
    return await ExecuteLoopAsync(instance, ct);
}

2. ExecuteLoopAsync — the main loop

This is the core. Implement exactly as specified below. Do not deviate
from the sequence.

csharpprivate async Task<ExecutionInstance> ExecuteLoopAsync(
    ExecutionInstance instance,
    CancellationToken hostCt)
{
    var claimantId = _options.ClaimantId;

    // ── Phase 1: Acquire claim ────────────────────────────────────────────────
    var claimed = await _lockProvider.TryAcquireAsync(
        instance.InstanceId, claimantId, _options.DefaultLeaseDuration, hostCt);

    if (!claimed)
    {
        _logger.LogDebug("Instance {Id} already claimed — skipping", instance.InstanceId);
        return instance;
    }

    // Register with lease renewal service
    var executionCts = CancellationTokenSource.CreateLinkedTokenSource(hostCt);
    ActiveExecutions[instance.InstanceId] = executionCts;
    var ct = executionCts.Token;

    try
    {
        // ── Phase 2: Load fresh state ──────────────────────────────────────────
        var fresh = await _stateStore.LoadAsync(instance.InstanceId, ct);
        if (fresh == null)
        {
            _logger.LogWarning("Instance {Id} not found after claim", instance.InstanceId);
            return instance;
        }
        instance = fresh;

        var definition = await _workflowRegistry.GetAsync(
            instance.WorkflowId, instance.WorkflowVersion, ct);
        if (definition is null)
        {
            await FaultInstanceAsync(instance, "DEFINITION_NOT_FOUND",
                $"Workflow '{instance.WorkflowId}@{instance.WorkflowVersion}' not found in registry.",
                instance.CurrentNodeId);
            await _stateStore.SaveAsync(instance, ct);
            EmitFinalEvent(instance);
            return instance;
        }

        // Use per-workflow lease duration override when specified
        var leaseDuration = definition.DefaultLeaseDuration ?? _options.DefaultLeaseDuration;

        // Update claim fields on the instance
        instance.ClaimedBy      = claimantId;
        instance.ClaimedAt      = DateTimeOffset.UtcNow;
        instance.ClaimExpiresAt = DateTimeOffset.UtcNow.Add(leaseDuration);

        // Dequeue any pending child signals (processed inside loop/parallel executors)
        await _stateStore.DequeueChildSignalsAsync(instance.InstanceId, ct);

    // ── Phase 3: Execution loop ───────────────────────────────────────────
    //
    // PERSISTENCE MODEL:
    // SaveAsync is called at loop EXIT only (Phase 4), not after every node.
    // A chain of synchronous nodes runs entirely in memory. On crash recovery
    // the sweep service reclaims the instance and the chain re-executes from
    // the last saved CurrentNodeId. All node executors MUST be idempotent.
    // The only exception is the checkpoint node, which sets ForceCheckpoint=true
    // in its NodeResult, triggering an immediate mid-loop SaveAsync (see step 3k).
    //
    var deadlineHandled = false;
    while (true)
    {
            ct.ThrowIfCancellationRequested();

            // ── 3a. Check instance-level deadline ─────────────────────────────
            if (!deadlineHandled && instance.Deadline.HasValue
                && instance.Deadline.Value <= DateTimeOffset.UtcNow)
            {
                if (definition.GlobalOnDeadlineNodeId != null)
                {
                    instance.CurrentNodeId = definition.GlobalOnDeadlineNodeId;
                    deadlineHandled = true; // prevent re-triggering on subsequent iterations
                    continue;
                }
                else
                {
                    instance.Status      = ExecStatus.TimedOut;
                    instance.CompletedAt = DateTimeOffset.UtcNow;
                    instance.Error = new ExecutionError
                    {
                        Code       = "INSTANCE_DEADLINE_EXCEEDED",
                        Message    = $"Instance deadline {instance.Deadline} exceeded",
                        NodeId     = instance.CurrentNodeId,
                        OccurredAt = DateTimeOffset.UtcNow
                    };
                    break;
                }
            }

            // ── 3b. Load current node ─────────────────────────────────────────
            if (!definition.Nodes.TryGetValue(instance.CurrentNodeId, out var node))
            {
                await FaultInstanceAsync(instance,
                    "NODE_NOT_FOUND",
                    $"Node '{instance.CurrentNodeId}' not found in workflow '{instance.WorkflowId}'",
                    instance.CurrentNodeId);
                break;
            }

            // ── 3c. MaxVisits guard ───────────────────────────────────────────
            var visits = instance.NodeVisitCounts.GetValueOrDefault(node.Id, 0) + 1;
            instance.NodeVisitCounts[node.Id] = visits;

            if (visits > node.MaxVisits)
            {
                instance.Status      = ExecStatus.CycleDetected;
                instance.CompletedAt = DateTimeOffset.UtcNow;
                instance.Error = new ExecutionError
                {
                    Code       = "MAX_VISITS_EXCEEDED",
                    Message    = $"Node '{node.Id}' visited {visits} times (max {node.MaxVisits})",
                    NodeId     = node.Id,
                    OccurredAt = DateTimeOffset.UtcNow
                };
                break;
            }

            // ── 3d. Execute node (with retry) ─────────────────────────────────
            EmitEvent("flow.node.started", instance, node);
            var startedAt = DateTimeOffset.UtcNow;

            var (result, attemptCount) = await ExecuteWithRetryAsync(
                node, instance, definition, ct);

            var completedAt = DateTimeOffset.UtcNow;
            EmitEvent("flow.node.completed", instance, node, result,
                (completedAt - startedAt).TotalMilliseconds);

            // ── 3e. Apply storeAs ─────────────────────────────────────────────
            if (node.StoreAs != null && result.Output != null)
                instance.Context.Store(node.StoreAs, result.Output);

            // ── 3f. Build history record ──────────────────────────────────────
            var record = new NodeExecutionRecord
            {
                NodeId         = node.Id,
                NodeType       = node.Type,
                NodeLabel      = node.Label,
                StartedAt      = startedAt,
                CompletedAt    = completedAt,
                Outcome        = result.Outcome,
                StoredAs       = node.StoreAs,
                OutputSnapshot = result.Output,
                HttpStatusCode = result.HttpStatusCode,
                AttemptNumber  = attemptCount,
                Error          = result.Error
            };

            // ── 3g. Handle suspension (wait / human nodes) ────────────────────
            if (result.PendingWait != null)
            {
                instance.PendingWait       = result.PendingWait;
                instance.PendingHumanTaskId= result.HumanTaskId;
                instance.Status            = ExecStatus.Suspended;
                instance.UpdatedAt         = DateTimeOffset.UtcNow;
                record.BranchTaken         = "suspended";
                instance.History.Add(record);

                EmitEvent("flow.instance.suspended", instance);
                break; // → Phase 4: save + release
            }

            // ── 3h. Resolve next node via edge evaluation ─────────────────────
            string? nextNodeId = null;
            string? branchLabel = null;

            foreach (var edge in node.Edges)
            {
                // Check outcome match (null/empty On = catch-all)
                var outcomeMatches = edge.On == null
                    || edge.On.Count == 0
                    || edge.On.Contains(result.Outcome);

                if (!outcomeMatches) continue;

                // Evaluate When predicate if present
                if (edge.When != null)
                {
                    var ctx = BuildExecutionContext(instance, definition);
                    var whenMet = await _whenEvaluator.EvaluateAsync(edge.When, ctx, ct);
                    if (!whenMet) continue;
                }

                nextNodeId  = edge.NextNodeId;
                branchLabel = edge.Label;
                break;
            }

            record.BranchTaken = branchLabel;
            instance.History.Add(record);

            // ── 3i. Handle no matching edge ───────────────────────────────────
            if (nextNodeId == null)
            {
                // No edges at all = terminal output node
                if (node.Edges.Count == 0)
                {
                    instance.Status      = ExecStatus.Completed;
                    instance.CompletedAt = DateTimeOffset.UtcNow;
                    instance.Output      = result.Output;
                    break;
                }

                // Error outcome with global handler
                if (result.Outcome == DecisionOutcome.Error
                    && definition.GlobalOnFailureNodeId != null)
                {
                    instance.CurrentNodeId = definition.GlobalOnFailureNodeId;
                    continue;
                }

                // No handler — fault
                await FaultInstanceAsync(instance,
                    "NO_MATCHING_EDGE",
                    $"No edge matched outcome '{result.Outcome}' from node '{node.Id}'",
                    node.Id);
                break;
            }

            // ── 3j. Advance to next node ──────────────────────────────────────
            instance.CurrentNodeId = nextNodeId;
            instance.UpdatedAt     = DateTimeOffset.UtcNow;

            // ── 3k. Checkpoint — force mid-loop save if node requested it ────────
            // Only the checkpoint node executor sets ForceCheckpoint = true.
            // All other executors leave it false — saves happen at loop exit only.
            // After a checkpoint save, CurrentNodeId is already nextNodeId so
            // crash recovery resumes from the node AFTER the checkpoint, not before.
            if (result.ForceCheckpoint)
            {
                _logger.LogDebug("Checkpoint at node {Id} instance {Inst}",
                    node.Id, instance.InstanceId);
                EmitEvent("flow.checkpoint.saved", instance);
                await SaveWithRetryAsync(instance, claimantId, ct);
                // Do NOT release claim — execution is still active
            }

            // continue loop
        }
    }
    catch (OperationCanceledException) when (!hostCt.IsCancellationRequested)
    {
        // Per-node timeout — already handled inside ExecuteWithRetryAsync
        // If we reach here the instance is already faulted
    }
    catch (OperationCanceledException)
    {
        // Host shutdown — save state as-is (sweep will reclaim)
        instance.UpdatedAt = DateTimeOffset.UtcNow;
        // Status stays Running — do NOT transition to Cancelled
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Unhandled exception in engine loop for {Id}", instance.InstanceId);
        await FaultInstanceAsync(instance, "ENGINE_ERROR", ex.Message,
            instance.CurrentNodeId);
    }
    finally
    {
        // ── Phase 4: Save and release ─────────────────────────────────────────
        ActiveExecutions.TryRemove(instance.InstanceId, out _);
        executionCts.Dispose();

        instance.UpdatedAt = DateTimeOffset.UtcNow;
        if (IsTerminal(instance.Status) && !instance.CompletedAt.HasValue)
            instance.CompletedAt = DateTimeOffset.UtcNow;

        await SaveWithRetryAsync(instance, claimantId, hostCt);

        if (instance.Status == ExecStatus.Suspended || IsTerminal(instance.Status))
            await _lockProvider.ReleaseAsync(instance.InstanceId, claimantId, hostCt);

        // Clear claim fields after release
        instance.ClaimedBy      = null;
        instance.ClaimedAt      = null;
        instance.ClaimExpiresAt = null;

        EmitFinalEvent(instance);

        // After suspension: flush any child signals enqueued while the parent's lock was
        // held (e.g. sequential loop items that completed synchronously before the parent
        // could suspend). Without this, those signals would be stuck until the sweep fires.
        if (instance.Status == ExecStatus.Suspended)
        {
            var pendingSignals = await _stateStore.DequeueChildSignalsAsync(
                instance.InstanceId, hostCt);

            if (pendingSignals.Count > 0)
            {
                foreach (var sig in pendingSignals)
                    await _stateStore.EnqueueChildSignalAsync(instance.InstanceId, sig, hostCt);

                var reloaded = await _stateStore.LoadAsync(instance.InstanceId, hostCt);
                if (reloaded?.Status == ExecStatus.Suspended)
                {
                    reloaded.Status    = ExecStatus.Running;
                    reloaded.UpdatedAt = DateTimeOffset.UtcNow;
                    _ = Task.Run(() => ExecuteLoopAsync(reloaded, hostCt), hostCt);
                }
            }
        }

        // If this instance has a parent, signal it
        if (IsTerminal(instance.Status) && instance.ParentInstanceId != null)
            await ChildSignalHandler.SignalParentAsync(instance, _stateStore,
                _lockProvider, this, _logger, hostCt);
    }

    return instance;
}

3. ExecuteWithRetryAsync

csharpprivate async Task<(NodeResult result, int attemptCount)> ExecuteWithRetryAsync(
    NodeDefinition    node,
    ExecutionInstance instance,
    WorkflowDefinition definition,
    CancellationToken ct)
{
    var policy       = node.RetryPolicy ?? definition.DefaultRetryPolicy;
    var executor     = _executorRegistry.Get(node.Type)
                       ?? throw new NodeExecutorNotFoundException(node.Type);
    var ctx          = BuildExecutionContext(instance, definition);
    var attemptCount = 0;

    while (true)
    {
        attemptCount++;

        // Apply per-node execution timeout
        using var nodeCts = node.ExecutionTimeout.HasValue
            ? CancellationTokenSource.CreateLinkedTokenSource(ct)
            : null;
        if (nodeCts != null)
            nodeCts.CancelAfter(node.ExecutionTimeout!.Value);
        var nodeCt = nodeCts?.Token ?? ct;

        NodeResult result;
        try
        {
            result = await executor.ExecuteAsync(node, instance, ctx, nodeCt);
        }
        catch (OperationCanceledException) when (
            nodeCts != null && nodeCts.IsCancellationRequested)
        {
            // Per-node timeout (not host cancellation)
            result = new NodeResult
            {
                Outcome = DecisionOutcome.Error,
                Error   = new ExecutionError
                {
                    Code       = "NODE_EXECUTION_TIMEOUT",
                    Message    = $"Node '{node.Id}' exceeded timeout {node.ExecutionTimeout}",
                    NodeId     = node.Id,
                    OccurredAt = DateTimeOffset.UtcNow
                }
            };
        }
        catch (Exception ex)
        {
            result = new NodeResult
            {
                Outcome = DecisionOutcome.Error,
                Error   = new ExecutionError
                {
                    Code       = "NODE_EXECUTOR_EXCEPTION",
                    Message    = ex.Message,
                    Detail     = ex.ToString(),
                    NodeId     = node.Id,
                    OccurredAt = DateTimeOffset.UtcNow
                }
            };
        }

        // Check if retry applies
        if (policy != null
            && policy.RetryOn.Contains(result.Outcome)
            && attemptCount < policy.MaxAttempts
            && !ct.IsCancellationRequested)
        {
            var delay = RetryHandler.CalculateDelay(policy, attemptCount);
            _logger.LogDebug(
                "Retrying node {NodeId} attempt {Attempt}/{Max} after {Delay}ms",
                node.Id, attemptCount, policy.MaxAttempts, delay.TotalMilliseconds);
            await Task.Delay(delay, ct);
            continue;
        }

        return (result, attemptCount);
    }
}

4. WhenEvaluator

Instance-based singleton. Scripts compiled once and cached up to FlowEngineOptions.ScriptCacheSize entries (default 1000). When the cache is full, 10% of entries are evicted to prevent unbounded memory growth. Shared across all concurrent workflow instances.

csharp/// <summary>
/// Evaluates EdgeDefinition.When predicate expressions using Roslyn scripting.
/// Exposes four dictionaries as script globals: Context, Entity, Params, Engine.
/// </summary>
public class WhenEvaluator
{

    {
        /// <summary>Workflow context dictionary ($.context.*) with unwrapped CLR values.</summary>
        public IReadOnlyDictionary<string, object?> Context { get; set; } = null!;
        /// <summary>
        /// Engine metadata: instanceId, workflowId, currentNodeId, status,
        /// visits (node visit count map), totalVisits.
        /// </summary>
        public IReadOnlyDictionary<string, object?> Engine  { get; set; } = null!;
        /// <summary>Primary entity ($.entity) as string-keyed CLR dictionary.</summary>
        public IReadOnlyDictionary<string, object?> Entity  { get; set; } = null!;
        /// <summary>Workflow start parameters ($.params.*).</summary>
        public IReadOnlyDictionary<string, object?> Params  { get; set; } = null!;
    }

    private readonly int _maxCacheSize;
    private readonly Lock _cacheLock = new();
    private readonly Dictionary<string, Script<object>> _cache = [];

    private static readonly ScriptOptions _scriptOptions = ScriptOptions.Default
        .AddImports("System", "System.Linq", "System.Collections.Generic", "System.Text.Json")
        .AddReferences(
            typeof(object).Assembly,
            typeof(Enumerable).Assembly,
            typeof(System.Text.Json.JsonElement).Assembly);

    public WhenEvaluator(FlowEngineOptions options)
    {
        _maxCacheSize = options.ScriptCacheSize > 0 ? options.ScriptCacheSize : 1000;
    }

    /// <summary>Evaluates a boolean edge-guard expression. Truthy = non-null, non-false, non-zero, non-empty.</summary>
    public async Task<bool> EvaluateAsync(
        string expression, IExecutionContext ctx, CancellationToken ct)
    {
        var globals = BuildGlobals(ctx);
        var script  = GetOrAddScript(expression);
        var state   = await script.RunAsync(globals, cancellationToken: ct);
        return IsTruthy(state.ReturnValue);
    }

    /// <summary>
    /// Evaluates an expression and returns its result as a string.
    /// Used by WaitNodeConfig.CorrelationKeyExpression to derive a dynamic correlation key.
    /// </summary>
    public async Task<string?> EvaluateAsStringAsync(
        string expression, IExecutionContext ctx, CancellationToken ct)
    {
        var globals = BuildGlobals(ctx);
        var script  = GetOrAddScript(expression);
        var state   = await script.RunAsync(globals, cancellationToken: ct);
        return state.ReturnValue?.ToString();
    }

    /// <summary>
    /// Evaluates an expression and returns the raw Roslyn result preserving its runtime type.
    /// Used by ComputeNodeExecutor to avoid string coercion before JSON serialisation.
    /// </summary>
    public async Task<object?> EvaluateAsObjectAsync(
        string expression, IExecutionContext ctx, CancellationToken ct)
    {
        var globals = BuildGlobals(ctx);
        var script  = GetOrAddScript(expression);
        var state   = await script.RunAsync(globals, cancellationToken: ct);
        return state.ReturnValue;
    }

    /// <summary>
    /// Gets or adds a compiled script from the bounded cache.
    /// Evicts 10% of entries when cache reaches _maxCacheSize.
    /// </summary>
    private Script<object> GetOrAddScript(string expression)
    {
        lock (_cacheLock)
        {
            if (_cache.TryGetValue(expression, out var cached))
                return cached;

            if (_cache.Count >= _maxCacheSize)
            {
                var toRemove = (int)(_cache.Count * 0.1);
                var keysToRemove = _cache.Keys.Take(toRemove).ToList();
                foreach (var key in keysToRemove)
                    _cache.Remove(key);
            }

            var script = CompileScript(expression);
            _cache[expression] = script;
            return script;
        }
    }

    private static Script<object> CompileScript(string expression)
    {
        // Wrap bare expressions in a return statement
        var code = expression.TrimStart().StartsWith("return ", StringComparison.Ordinal)
            ? expression
            : $"return (object)({expression});";
        return CSharpScript.Create<object>(code, _scriptOptions, typeof(Globals));
    }

    private static bool IsTruthy(object? value) => value switch
    {
        null           => false,
        bool b         => b,
        int i          => i != 0,
        string s       => !string.IsNullOrEmpty(s),
        _              => true
    };

    private static Globals BuildGlobals(IExecutionContext ctx)
    {
        var engine = new Dictionary<string, object?>
        {
            ["instanceId"]    = ctx.Instance.InstanceId,
            ["workflowId"]    = ctx.Instance.WorkflowId,
            ["currentNodeId"] = ctx.Instance.CurrentNodeId,
            ["status"]        = ctx.Instance.Status.ToString(),
            ["visits"]        = new Dictionary<string, int>(ctx.Instance.NodeVisitCounts),
            ["totalVisits"]   = ctx.TotalVisitCount
        };

        return new Globals
        {
            Context = UnwrapContext(ctx.Instance.Context.Context),
            Engine  = engine,
            Entity  = UnwrapContextValue(ctx.Instance.Context.Entity),
            Params  = UnwrapContext(ctx.Instance.Context.Params ?? new())
        };
    }

    private static IReadOnlyDictionary<string, object?> UnwrapContext(
        IDictionary<string, ContextValue> source)
        => source.ToDictionary(kv => kv.Key, kv => UnwrapValue(kv.Value));

    private static IReadOnlyDictionary<string, object?> UnwrapContextValue(ContextValue? v)
    {
        if (v is JsonContextValue jv)
            return jv.Value.Deserialize<Dictionary<string, object?>>()
                   ?? new Dictionary<string, object?>();
        return new Dictionary<string, object?>();
    }

    private static object? UnwrapValue(ContextValue? v) => v switch
    {
        JsonContextValue     j => j.Value.Deserialize<object?>(),
        NullContextValue       => null,
        DocumentContextValue d => new { d.StoreRef, d.FileName, d.ContentType },
        null                   => null,
        _                      => null
    };
}

5. RetryHandler

csharppublic static class RetryHandler
{
    private static readonly Random _jitter = new();

    public static TimeSpan CalculateDelay(RetryPolicy policy, int attemptNumber)
    {
        // attemptNumber is 1-based; first retry fires after attempt 1
        var attempt = attemptNumber - 1; // 0-based exponent

        var delay = policy.Backoff switch
        {
            BackoffType.Fixed       => policy.InitialDelay,
            BackoffType.Linear      => policy.InitialDelay * attemptNumber,
            BackoffType.Exponential => policy.InitialDelay * Math.Pow(2, attempt),
            BackoffType.ExponentialWithJitter =>
                policy.InitialDelay * Math.Pow(2, attempt)
                + TimeSpan.FromMilliseconds(_jitter.Next(0, 1000)),
            _ => policy.InitialDelay
        };

        // Cap at MaxDelay
        return delay > policy.MaxDelay ? policy.MaxDelay : delay;
    }
}

6. ChildSignalHandler

csharppublic static class ChildSignalHandler
{
    /// <summary>
    /// Called when a child instance (loop item, sub-workflow, parallel branch)
    /// reaches a terminal state. Signals the parent to check if it can resume.
    /// </summary>
    public static async Task SignalParentAsync(
        ExecutionInstance         child,
        IExecutionStateStore      stateStore,
        IDistributedLockProvider  lockProvider,
        FlowEngine                engine,
        ILogger                   logger,
        CancellationToken         ct)
    {
        var signal = new ChildSignal
        {
            ParentNodeId      = child.ParentNodeId!,
            ChildInstanceId   = child.InstanceId,
            ChildStatus       = child.Status,
            ChildOutput       = child.Output,
            OccurredAt        = DateTimeOffset.UtcNow
        };

        // Try to directly resume the parent if unclaimed
        var claimantId = $"child-signal-{child.InstanceId}";
        var claimed = await lockProvider.TryAcquireAsync(
            child.ParentInstanceId!, claimantId,
            TimeSpan.FromSeconds(30), ct);

        if (claimed)
        {
            // We got the parent — resume it directly
            // Pass claimantId so ResumeChildAsync can release this lock before re-acquiring
            await engine.ResumeChildAsync(
                child.ParentInstanceId!, signal, claimantId, ct);
        }
        else
        {
            // Parent is claimed by another server processing it.
            // Enqueue the signal — parent will pick it up on next save/load cycle.
            await stateStore.EnqueueChildSignalAsync(
                child.ParentInstanceId!, signal, ct);
            logger.LogDebug(
                "Child signal queued for parent {ParentId} node {NodeId}",
                child.ParentInstanceId, child.ParentNodeId);
        }
    }
}

7. ExecutionSweepService

csharppublic class ExecutionSweepService : BackgroundService
{
    private readonly IDistributedLockProvider _lockProvider;
    private readonly FlowEngine               _engine;
    private readonly FlowEngineOptions        _options;
    private readonly ILogger<ExecutionSweepService> _logger;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        // Stagger startup to avoid all servers sweeping simultaneously
        await Task.Delay(TimeSpan.FromSeconds(Random.Shared.Next(1, 15)), ct);

        while (!ct.IsCancellationRequested)
        {
            try
            {
                var resumable = await _lockProvider.FindResumableInstanceIdsAsync(
                    DateTimeOffset.UtcNow,
                    _options.SweepBatchSize,
                    ct);

                foreach (var instanceId in resumable)
                {
                    // Fire and forget — each resumption runs independently
                    _ = Task.Run(() => _engine.ResumeFromSweepAsync(instanceId, ct), ct);
                }

                if (resumable.Count > 0)
                    _logger.LogInformation("Sweep found {Count} resumable instances",
                        resumable.Count);
            }
            catch (Exception ex) when (!ct.IsCancellationRequested)
            {
                _logger.LogError(ex, "Error during execution sweep");
            }

            await Task.Delay(_options.SweepInterval, ct);
        }
    }
}

8. LeaseRenewalService

Purpose: Prevents ExecutionSweepService on other nodes from reclaiming instances that are still actively executing locally (false “orphaned” detection).
Interval: FlowEngineOptions.LeaseRenewalInterval (default 13s).
Constraint: LeaseRenewalInterval < DefaultLeaseDuration / 2 (default: 13s < 15s ✓).
Failure mode: If RenewAsync returns false (another node claimed the instance), the local CancellationTokenSource for that execution is cancelled immediately. 

csharppublic class LeaseRenewalService : BackgroundService
{
    private readonly IDistributedLockProvider _lockProvider;
    private readonly FlowEngine               _engine;
    private readonly FlowEngineOptions        _options;
    private readonly ILogger<LeaseRenewalService> _logger;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            await Task.Delay(_options.LeaseRenewalInterval, ct);

            foreach (var (instanceId, executionCts) in _engine.ActiveExecutions)
            {
                try
                {
                    var renewed = await _lockProvider.RenewAsync(
                        instanceId,
                        _options.ClaimantId,
                        _options.DefaultLeaseDuration,
                        ct);

                    if (!renewed)
                    {
                        _logger.LogCritical(
                            "Failed to renew lease for {InstanceId} — " +
                            "claim may have been stolen. Aborting execution.",
                            instanceId);
                        // Cancel the execution — engine loop will save state and exit
                        executionCts.Cancel();
                    }
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Lease renewal error for {InstanceId}", instanceId);
                }
            }
        }
    }
}

11. MigrateInstanceAsync / DeleteWorkflowVersionAsync

csharp/// <summary>
/// Migrate a non-terminal instance to a different workflow version.
/// nodeIdMap remaps CurrentNodeId when node IDs changed between versions.
/// Throws if the target node does not exist in the target definition.
/// Appends a "migration" history record.
/// </summary>
public async Task<ExecutionInstance> MigrateInstanceAsync(
    string                               instanceId,
    string                               targetVersion,
    IReadOnlyDictionary<string, string>? nodeIdMap = null,
    CancellationToken                    ct         = default)
{
    var instance = await _stateStore.LoadAsync(instanceId, ct)
        ?? throw new InstanceNotFoundException(instanceId);

    if (IsTerminal(instance.Status))
        throw new InvalidOperationException(
            $"Instance '{instanceId}' is in terminal state {instance.Status} and cannot be migrated.");

    var targetDef = await _workflowRegistry.GetAsync(instance.WorkflowId, targetVersion, ct)
        ?? throw new InvalidOperationException(
            $"Workflow '{instance.WorkflowId}@{targetVersion}' not found in registry.");

    var resolvedNodeId = nodeIdMap != null
        && nodeIdMap.TryGetValue(instance.CurrentNodeId, out var mapped)
            ? mapped
            : instance.CurrentNodeId;

    if (!targetDef.Nodes.ContainsKey(resolvedNodeId))
        throw new InvalidOperationException(
            $"Node '{resolvedNodeId}' does not exist in '{instance.WorkflowId}@{targetVersion}'. "
            + "Provide a nodeIdMap entry to remap it.");

    var fromVersion = instance.WorkflowVersion;
    instance.WorkflowVersion = targetVersion;
    instance.CurrentNodeId   = resolvedNodeId;
    instance.UpdatedAt       = DateTimeOffset.UtcNow;

    instance.History.Add(new NodeExecutionRecord
    {
        NodeId      = resolvedNodeId,
        NodeType    = "migration",
        NodeLabel   = $"Migrated {fromVersion} → {targetVersion}",
        StartedAt   = instance.UpdatedAt,
        CompletedAt = instance.UpdatedAt,
        Outcome     = DecisionOutcome.Match
    });

    await _stateStore.SaveAsync(instance, ct);
    return instance;
}

/// <summary>
/// Delete a workflow version. Throws if any Running or Suspended instances exist.
/// Transitions Active → Draft first (registry requires Draft to allow deletion).
/// </summary>
public async Task DeleteWorkflowVersionAsync(
    string workflowId, string version, CancellationToken ct = default)
{
    // Block on Running
    var running = await _stateStore.QueryAsync(new ExecutionQuery
    {
        WorkflowId = workflowId, WorkflowVersion = version,
        Status = ExecStatus.Running, Take = 1
    }, ct);
    if (running.TotalCount > 0)
        throw new InvalidOperationException(
            $"Workflow '{workflowId}@{version}' has {running.TotalCount} running instance(s). Cannot delete.");

    // Block on Suspended
    var suspended = await _stateStore.QueryAsync(new ExecutionQuery
    {
        WorkflowId = workflowId, WorkflowVersion = version,
        Status = ExecStatus.Suspended, Take = 1
    }, ct);
    if (suspended.TotalCount > 0)
        throw new InvalidOperationException(
            $"Workflow '{workflowId}@{version}' has {suspended.TotalCount} suspended instance(s). "
            + "Use MigrateInstanceAsync to move them to a different version first.");

    // Transition Active → Draft so registry allows deletion
    var def = await _workflowRegistry.GetAsync(workflowId, version, ct);
    if (def?.Status == DefinitionStatus.Active)
        await _workflowRegistry.TransitionStatusAsync(workflowId, version, DefinitionStatus.Draft, ct);

    await _workflowRegistry.DeleteAsync(workflowId, version, ct);
}

14. CancelAsync — implementation spec

csharp/// <summary>
/// Cancel a running or suspended instance. Transitions to Cancelled (terminal).
/// Closes any open HumanTasks. Cascades cancellation to all non-terminal child instances.
/// </summary>
public async Task<ExecutionInstance> CancelAsync(
    string instanceId, string reason, CancellationToken ct)
{
    // 1. Try to claim the instance (prevents race with active execution)
    var claimed = await _lockProvider.TryAcquireAsync(
        instanceId, _options.ClaimantId, _options.DefaultLeaseDuration, ct);

    try
    {
        var instance = await _stateStore.LoadAsync(instanceId, ct)
            ?? throw new InstanceNotFoundException(instanceId);

        // 2. If already terminal, return as-is (idempotent)
        if (IsTerminal(instance.Status))
            return instance;

        // 3. Transition to Cancelled
        instance.Status      = ExecStatus.Cancelled;
        instance.CompletedAt = DateTimeOffset.UtcNow;
        instance.UpdatedAt   = DateTimeOffset.UtcNow;
        instance.Error       = new ExecutionError
        {
            Code       = "CANCELLED",
            Message    = reason,
            NodeId     = instance.CurrentNodeId,
            OccurredAt = DateTimeOffset.UtcNow
        };

        // 4. Close any open HumanTasks for this instance
        var taskStore = _services.GetService<IHumanTaskStore>();
        if (taskStore != null)
        {
            var openTasks = await taskStore.GetByInstanceAsync(instanceId, ct);
            foreach (var task in openTasks.Where(t =>
                t.Status is HumanTaskStatus.Open or HumanTaskStatus.InProgress))
            {
                task.Status = HumanTaskStatus.Cancelled;
                await taskStore.SaveAsync(task, ct);
            }
        }

        // 5. Clear pending wait so sweep service ignores this instance
        instance.PendingWait = null;

        // 6. Save, emit, then cascade cancel to children (best-effort)
        await SaveWithRetryAsync(instance, _options.ClaimantId, ct);
        EmitFinalEvent(instance);

        var children = await _stateStore.GetChildrenAsync(instanceId, ct);
        foreach (var child in children.Where(c => !IsTerminal(c.Status)))
        {
            try { await CancelAsync(child.InstanceId, "Parent workflow cancelled", ct); }
            catch { /* swallow — best-effort cascade */ }
        }

        return instance;
    }
    finally
    {
        if (claimed) await _lockProvider.ReleaseAsync(instanceId, _options.ClaimantId, ct);
    }
}

15. HumanNodeExecutor — full behavior spec

The human node executor is the most stateful executor — it creates a HumanTask,
suspends the instance, and must handle re-execution on resumption without
creating duplicate tasks.

csharp/// <summary>
/// Human node executor. Creates a HumanTask and suspends the instance.
/// On re-execution after resumption (sweep or direct resume), detects
/// the existing task by (InstanceId, NodeId) and returns Match if quorum
/// has been reached, or re-suspends if not yet reached.
///
/// Idempotency: if a HumanTask already exists for this (InstanceId, NodeId),
/// the executor MUST NOT create a new one. This handles crash-recovery
/// re-execution correctly.
/// </summary>
public sealed class HumanNodeExecutor : INodeExecutor
{
    public string NodeType => "human";

    private readonly IHumanTaskStore _taskStore;
    private readonly IFlowEngineTelemetry _telemetry;

    public async Task<NodeResult> ExecuteAsync(
        NodeDefinition node, ExecutionInstance instance,
        IExecutionContext ctx, CancellationToken ct)
    {
        var config = node.Config.Deserialize<HumanNodeConfig>()!;

        // ── Step 1: Check if a task already exists for this node ─────────────
        // Handles crash-recovery re-execution — do not create a duplicate task.
        var existingTasks = await _taskStore.GetByInstanceAsync(instance.InstanceId, ct);
        var existing = existingTasks.FirstOrDefault(t => t.NodeId == node.Id);

        if (existing != null)
        {
            // Task exists — check if quorum has been reached
            if (existing.Status == HumanTaskStatus.Completed
                && existing.Responses.Count >= existing.Quorum)
            {
                // Quorum reached — write response to context and return Match
                WriteResponseToContext(existing, instance);
                return new NodeResult { Outcome = DecisionOutcome.Match };
            }

            // Task exists but quorum not yet met — re-suspend with same wait descriptor
            return new NodeResult
            {
                Outcome     = DecisionOutcome.Match,
                PendingWait = BuildWaitDescriptor(existing, instance)
            };
        }

        // ── Step 2: Resolve dynamic fields from context ───────────────────────
        var title      = InterpolateTemplate(config.Title, ctx);
        var assignedTo = ResolveAssignedTo(config.AssignedTo, ctx);
        var dueAt      = config.DueIn != null
            ? await ctx.ResolveDateTimeAsync(config.DueIn, ct)
            : (DateTimeOffset?)null;

        // ── Step 3: Create the HumanTask ──────────────────────────────────────
        var taskId = Guid.NewGuid().ToString("N");
        var task   = new HumanTask
        {
            TaskId      = taskId,
            InstanceId  = instance.InstanceId,
            NodeId      = node.Id,
            WorkflowId  = instance.WorkflowId,
            Title       = title,
            AssignedTo  = assignedTo,
            Channel     = config.Channel,
            Notification= config.Notification,
            FormSchema  = config.FormSchema,
            VisibleContext = config.VisibleContext ?? new(),
            Quorum      = config.Quorum,
            Status      = HumanTaskStatus.Open,
            CreatedAt   = DateTimeOffset.UtcNow,
            DueAt       = dueAt,
            Escalation  = config.Escalation,
            ContextSnapshot = BuildContextSnapshot(config.VisibleContext, ctx)
        };

        await _taskStore.SaveAsync(task, ct);

        // ── Step 4: If External channel, generate response token ──────────────
        if (config.Channel == TaskChannel.External && config.Notification != null)
        {
            var token = await _taskStore.GenerateResponseTokenAsync(taskId, ct);
            // Token embedded in notification link by the notification service
            _telemetry?.Emit("flow.task.created", new Dictionary<string, object?>
            {
                ["taskId"]      = taskId,
                ["instanceId"]  = instance.InstanceId,
                ["assignedTo"]  = assignedTo,
                ["channel"]     = config.Channel.ToString(),
                ["responseToken"] = token   // notification service reads this
            });
        }
        else
        {
            _telemetry?.Emit("flow.task.created", new Dictionary<string, object?>
            {
                ["taskId"]     = taskId,
                ["instanceId"] = instance.InstanceId,
                ["assignedTo"] = assignedTo,
                ["channel"]    = config.Channel.ToString()
            });
        }

        // ── Step 5: Build WaitDescriptor and suspend ──────────────────────────
        // Event name convention: "humantask.{taskId}" — SubmitResponseAsync
        // calls engine.ResumeByInstanceIdAsync which bypasses event lookup.
        // The WaitDescriptor here is purely informational for the sweep service
        // to detect overdue tasks.
        var waitDescriptor = new WaitDescriptor
        {
            EventName      = $"humantask.{taskId}",
            CorrelationKey = instance.InstanceId,
            TimeoutAt      = dueAt,            // sweep detects overdue via this
            OnTimeoutNodeId= null,             // escalation handled by task store sweep
            MinSignalCount = 1
        };

        return new NodeResult
        {
            Outcome      = DecisionOutcome.Match,
            PendingWait  = waitDescriptor,
            HumanTaskId  = taskId
        };
    }

    private static void WriteResponseToContext(HumanTask task, ExecutionInstance instance)
    {
        // Merge all quorum responses into context
        // For Quorum=1: single response under _humanTaskResponse
        // For Quorum>1: array under _humanTaskResponses + consensus enum
        if (task.Quorum == 1 && task.Responses.Count == 1)
        {
            instance.Context.Store("_humanTaskResponse",
                new JsonContextValue { Value = task.Responses[0].Values });
        }
        else
        {
            var allValues = task.Responses.Select(r => r.Values).ToList();
            instance.Context.Store("_humanTaskResponses",
                new JsonContextValue { Value = JsonSerializer.SerializeToElement(allValues) });
            // Also set _humanTaskResponse to first response for simple edge checks
            if (task.Responses.Count > 0)
                instance.Context.Store("_humanTaskResponse",
                    new JsonContextValue { Value = task.Responses[0].Values });

            // Compute consensus key for clean edge when predicates
            // Values: "unanimous", "majority", "any-denied", "split"
            var decisions = task.Responses
                .Select(r => r.Values.TryGetProperty("decision", out var d)
                    ? d.GetString() : null)
                .Where(d => d != null).ToList();

            string consensus;
            if (decisions.Count == 0)
            {
                consensus = "split";
            }
            else if (decisions.All(d => d == decisions[0]))
            {
                consensus = "unanimous";
            }
            else if (decisions.Any(d => d == "denied" || d == "rejected"))
            {
                consensus = "any-denied";
            }
            else
            {
                var majorityDecision = decisions
                    .GroupBy(x => x)
                    .OrderByDescending(g => g.Count())
                    .First();
                consensus = majorityDecision.Count() > task.Quorum / 2
                    ? "majority" : "split";
            }

            instance.Context.Store("_humanTaskConsensus",
                new JsonContextValue { Value = JsonSerializer.SerializeToElement(consensus) });
        }
    }

    private static string InterpolateTemplate(string template, IExecutionContext ctx)
    {
        // Replace {{$.context.x}} and {{$.entity.x}} with resolved values
        return System.Text.RegularExpressions.Regex.Replace(
            template, @"\{\{(\$\.[^}]+)\}\}", m =>
            {
                var val = ctx.Resolve(m.Groups[1].Value);
                return val is JsonContextValue jv ? jv.Value.ToString() : "";
            });
    }

    private static string ResolveAssignedTo(string assignedTo, IExecutionContext ctx)
    {
        // Step 1: Apply template interpolation FIRST
        // Handles patterns like "role:{{$.context.requiredCounselTier}}-counsel"
        var interpolated = InterpolateTemplate(assignedTo, ctx);

        // Step 2: If the interpolated result is a JSONPath, resolve it
        if (interpolated.StartsWith("$."))
            return ctx.ResolveAs<string>(interpolated) ?? interpolated;

        return interpolated;
    }

    private static WaitDescriptor BuildWaitDescriptor(HumanTask task, ExecutionInstance instance)
        => new WaitDescriptor
        {
            EventName      = $"humantask.{task.TaskId}",
            CorrelationKey = instance.InstanceId,
            TimeoutAt      = task.DueAt,
            MinSignalCount = 1
        };

    private static JsonElement BuildContextSnapshot(
        List<string>? paths, IExecutionContext ctx)
    {
        if (paths == null || paths.Count == 0)
            return JsonSerializer.SerializeToElement(new {});
        var dict = paths.ToDictionary(
            p => p,
            p => ctx.Resolve(p) is JsonContextValue jv
                    ? (object?)jv.Value.Deserialize<object>() : null);
        return JsonSerializer.SerializeToElement(dict);
    }
}

csharp/// <summary>
/// Local filesystem state store for development and testing.
/// Stores each ExecutionInstance as a JSON file named {instanceId}.json
/// in the configured directory. NOT safe for multi-server deployments.
///
/// Paired with InMemory lock provider only.
/// No infrastructure required — works immediately on any developer machine.
/// </summary>
public sealed class FileExecutionStateStore : IExecutionStateStore
{
    private readonly string _directory;
    private readonly SemaphoreSlim _lock = new(1, 1); // single-process file safety

    public FileExecutionStateStore(string directory)
    {
        _directory = directory;
        Directory.CreateDirectory(directory);
    }

    // ── Core CRUD ──────────────────────────────────────────────────────────────

    public async Task<ExecutionInstance?> LoadAsync(string instanceId, CancellationToken ct)
    {
        var path = FilePath(instanceId);
        if (!File.Exists(path)) return null;
        var json = await File.ReadAllTextAsync(path, ct);
        return JsonSerializer.Deserialize<ExecutionInstance>(json);
    }

    public async Task SaveAsync(ExecutionInstance instance, CancellationToken ct)
    {
        // Optimistic concurrency: read current version, fail if mismatch
        await _lock.WaitAsync(ct);
        try
        {
            var existing = await LoadAsync(instance.InstanceId, ct);
            if (existing != null && existing.Version != instance.Version - 1)
                throw new ConcurrencyException(instance.InstanceId);
            var json = JsonSerializer.Serialize(instance,
                new JsonSerializerOptions { WriteIndented = true });
            await File.WriteAllTextAsync(FilePath(instance.InstanceId), json, ct);
        }
        finally { _lock.Release(); }
    }

    public Task DeleteAsync(string instanceId, CancellationToken ct)
    {
        var path = FilePath(instanceId);
        if (File.Exists(path)) File.Delete(path);
        return Task.CompletedTask;
    }

    // ── Signal lookup ──────────────────────────────────────────────────────────

    public async Task<ExecutionInstance?> FindBySuspendedEventAsync(
        string eventName, string correlationKey, CancellationToken ct)
    {
        // Linear scan — acceptable for dev/test with small instance counts
        foreach (var file in Directory.GetFiles(_directory, "*.json"))
        {
            var inst = JsonSerializer.Deserialize<ExecutionInstance>(
                await File.ReadAllTextAsync(file, ct));
            if (inst?.Status == ExecStatus.Suspended
                && inst.PendingWait?.EventName == eventName
                && inst.PendingWait?.CorrelationKey == correlationKey)
                return inst;
        }
        return null;
    }

    public async Task<IReadOnlyList<ExecutionInstance>> FindTimedOutInstancesAsync(
        DateTimeOffset cutoff, CancellationToken ct)
    {
        var results = new List<ExecutionInstance>();
        foreach (var file in Directory.GetFiles(_directory, "*.json"))
        {
            var inst = JsonSerializer.Deserialize<ExecutionInstance>(
                await File.ReadAllTextAsync(file, ct));
            if (inst?.Status == ExecStatus.Suspended
                && inst.PendingWait?.TimeoutAt <= cutoff)
                results.Add(inst);
        }
        return results;
    }

    // ── Queries ────────────────────────────────────────────────────────────────

    public async Task<ExecutionQueryResult> QueryAsync(
        ExecutionQuery query, CancellationToken ct)
    {
        var all = new List<ExecutionInstance>();
        foreach (var file in Directory.GetFiles(_directory, "*.json"))
        {
            var inst = JsonSerializer.Deserialize<ExecutionInstance>(
                await File.ReadAllTextAsync(file, ct));
            if (inst == null) continue;
            if (query.WorkflowId    != null && inst.WorkflowId    != query.WorkflowId)    continue;
            if (query.CorrelationId != null && inst.CorrelationId != query.CorrelationId) continue;
            if (query.Status        != null && inst.Status        != query.Status)        continue;
            all.Add(inst);
        }
        all = all.OrderByDescending(i => i.UpdatedAt).ToList();
        var page = all.Skip(query.Skip).Take(query.Take).ToList();
        return new ExecutionQueryResult
        {
            Items      = page,
            TotalCount = all.Count,
            Skip       = query.Skip,
            Take       = query.Take
        };
    }

    public async Task<IReadOnlyList<ExecutionInstance>> GetChildrenAsync(
        string parentInstanceId, CancellationToken ct)
    {
        var results = new List<ExecutionInstance>();
        foreach (var file in Directory.GetFiles(_directory, "*.json"))
        {
            var inst = JsonSerializer.Deserialize<ExecutionInstance>(
                await File.ReadAllTextAsync(file, ct));
            if (inst?.ParentInstanceId == parentInstanceId)
                results.Add(inst);
        }
        return results;
    }

    // ── Child signals (stored in a separate sidecar file) ─────────────────────

    public async Task EnqueueChildSignalAsync(
        string parentInstanceId, ChildSignal signal, CancellationToken ct)
    {
        var path    = SignalPath(parentInstanceId);
        var signals = File.Exists(path)
            ? JsonSerializer.Deserialize<List<ChildSignal>>(
                await File.ReadAllTextAsync(path, ct)) ?? new()
            : new List<ChildSignal>();
        signals.Add(signal);
        await File.WriteAllTextAsync(path,
            JsonSerializer.Serialize(signals), ct);
    }

    public async Task<IReadOnlyList<ChildSignal>> DequeueChildSignalsAsync(
        string parentInstanceId, CancellationToken ct)
    {
        var path = SignalPath(parentInstanceId);
        if (!File.Exists(path)) return Array.Empty<ChildSignal>();
        var signals = JsonSerializer.Deserialize<List<ChildSignal>>(
            await File.ReadAllTextAsync(path, ct)) ?? new();
        File.Delete(path);   // atomic dequeue for single-process use
        return signals;
    }

    // ── Helpers ────────────────────────────────────────────────────────────────

    private string FilePath(string instanceId)
        => Path.Combine(_directory, $"{instanceId}.json");

    private string SignalPath(string instanceId)
        => Path.Combine(_directory, $"{instanceId}.signals.json");
}

17. TimerNodeExecutor

csharp/// <summary>
/// Suspends execution for a pure time delay. No external signal required.
/// The sweep service resumes the instance when WaitDescriptor.TimeoutAt fires.
///
/// Key distinction from WaitNodeExecutor:
///   wait node  — timeout emits Error (external signal didn't arrive in time)
///   timer node — timeout IS the happy path; always emits Match after delay
///
/// The timer executor builds a WaitDescriptor with a synthetic event name
/// that only the sweep service will ever match. The TimeoutAt IS the resume
/// trigger — no ResumeAsync call needed from harness code.
/// </summary>
public sealed class TimerNodeExecutor : INodeExecutor
{
    public string NodeType => "timer";

    private readonly ICalendarProvider      _calendar;
    private readonly IFlowEngineTelemetry   _telemetry;

    public TimerNodeExecutor(
        ICalendarProvider    calendar,
        IFlowEngineTelemetry telemetry)
    {
        _calendar  = calendar;
        _telemetry = telemetry;
    }

    public async Task<NodeResult> ExecuteAsync(
        NodeDefinition    node,
        ExecutionInstance instance,
        IExecutionContext  ctx,
        CancellationToken ct)
    {
        var config = node.Config.Deserialize<TimerNodeConfig>()
            ?? throw new InvalidOperationException(
                $"Timer node '{node.Id}' has missing or invalid config.");

        // Resolve the delay to an absolute DateTimeOffset via ICalendarProvider
        var resumeAt = await _calendar.ResolveAsync(
            config.Delay, DateTimeOffset.UtcNow, ct);

        // Synthetic event name — unique per instance+node so no collision across
        // multiple timer nodes in the same workflow or parallel branches
        var syntheticEvent = $"timer.{instance.InstanceId}.{node.Id}";

        _telemetry.Emit("flow.node.timer.suspended", new()
        {
            ["instanceId"] = instance.InstanceId,
            ["nodeId"]     = node.Id,
            ["delay"]      = config.Delay,
            ["resumeAt"]   = resumeAt.ToString("O")
        });

        // Return PendingWait — engine saves state and suspends
        // The sweep service calls ResumeFromSweepAsync when TimeoutAt <= now
        // Because TimeoutAt is the DESIRED resume time (not an error deadline),
        // the sweep service must emit Match (not Error) on timer resumption.
        // This is handled by ResumeFromSweepAsync: it checks if the suspended
        // node type is "timer" and passes outcome = Match instead of Error.
        return new NodeResult
        {
            Outcome     = DecisionOutcome.Match,
            PendingWait = new WaitDescriptor
            {
                EventName      = syntheticEvent,
                CorrelationKey = instance.InstanceId,  // unique — no collision
                TimeoutAt      = resumeAt,
                MinSignalCount = 1
                // OnTimeoutNodeId intentionally null — timer always succeeds
            }
        };
    }
}

Timer vs Wait — sweep service distinction

The sweep service's ResumeFromSweepAsync must distinguish timer timeout
(always Match) from wait timeout (Error). It does this by checking the node type:

csharp// In ResumeFromSweepAsync (or the sweep loop):
var definition = await _workflowRegistry.GetAsync(
    instance.WorkflowId, instance.WorkflowVersion, ct);
var currentNode = definition.Nodes[instance.CurrentNodeId];

var isTimerTimeout = currentNode.Type == "timer"
    && instance.PendingWait?.TimeoutAt <= DateTimeOffset.UtcNow;

if (isTimerTimeout)
{
    // Timer delay elapsed — Match outcome, clear wait, advance
    instance.PendingWait = null;
    instance.Status      = ExecStatus.Running;
    // signal context: no payload written (timer has no signal data)
}
else
{
    // Wait node timeout — Error outcome, write to context, take Error edge
    instance.Context.Store("_timedOut", new JsonContextValue
        { Value = JsonSerializer.SerializeToElement(true) });
    // engine loop takes the Error edge
}

csharp/// <summary>
/// Concrete implementation of IExecutionContext. Constructed per-node-execution
/// by the engine loop via BuildExecutionContext(). Scoped to a single node run.
/// </summary>
internal sealed class ExecutionContext : IExecutionContext
{
    private readonly WorkflowDefinition _definition;
    private readonly IServiceProvider   _services;
    private readonly ICalendarProvider  _calendar;

    public ExecutionInstance Instance { get; }
    public IServiceProvider  Services => _services;
    public ILogger Logger => _services.GetRequiredService<ILogger<ExecutionContext>>();
    public int TotalVisitCount => Instance.NodeVisitCounts.Values.Sum();
    public int GetVisitCount(string nodeId)
        => Instance.NodeVisitCounts.GetValueOrDefault(nodeId, 0);

    public ContextValue? Resolve(string jsonPath)
    {
        if (string.IsNullOrWhiteSpace(jsonPath))
            throw new ArgumentException("Path cannot be empty", nameof(jsonPath));

        return jsonPath switch
        {
            var p when p == "$.engine"
                => SerialiseEngine(),
            var p when p.StartsWith("$.engine.visits.")
                => ResolveVisit(p["$.engine.visits.".Length..]),
            var p when p.StartsWith("$.engine.")
                => ResolveEngineMeta(p["$.engine.".Length..]),
            "$.entity"
                => Instance.Context.Entity,
            var p when p.StartsWith("$.entity")
                => ResolveJsonPath(Instance.Context.Entity, p["$.entity".Length..]),
            var p when p.StartsWith("$.params.")
                => ResolveFromDict(Instance.Context.Params, p["$.params.".Length..]),
            var p when p.StartsWith("$.context.")
                => ResolveFromDict(Instance.Context.Context, p["$.context.".Length..]),
            _ => throw new ArgumentException(
                $"Unrecognised namespace: '{jsonPath}'. " +
                "Valid: $.entity, $.params, $.context, $.engine")
        };
    }

    public T? ResolveAs<T>(string jsonPath)
    {
        var value = Resolve(jsonPath);
        if (value is null or NullContextValue) return default;
        if (value is JsonContextValue jv) return jv.Value.Deserialize<T>();
        throw new InvalidCastException(
            $"Cannot convert {value.GetType().Name} to {typeof(T).Name}");
    }

    public void Store(string key, ContextValue value)
    {
        if (key.StartsWith("$.engine"))
            throw new InvalidOperationException("$.engine namespace is read-only.");
        Instance.Context.Store(key, value);
    }

    public Task<Credential?> GetCredentialAsync(string credentialRef, CancellationToken ct)
        => _services.GetRequiredService<ICredentialStore>().GetAsync(credentialRef, ct);

    public Task<Stream> GetDocumentAsync(DocumentContextValue doc, CancellationToken ct)
        => _services.GetRequiredService<IDocumentStore>().OpenReadAsync(doc.StoreRef, ct);

    public Task<DateTimeOffset> ResolveDateTimeAsync(string expression, CancellationToken ct)
        => _calendar.ResolveAsync(expression, DateTimeOffset.UtcNow, ct);

    // ── Private helpers ───────────────────────────────────────────────────────

    private ContextValue ResolveVisit(string nodeId)
    {
        var count = nodeId == "_total" ? TotalVisitCount
                    : Instance.NodeVisitCounts.GetValueOrDefault(nodeId, 0);
        return new JsonContextValue { Value = JsonSerializer.SerializeToElement(count) };
    }

    private ContextValue? ResolveEngineMeta(string key) => key switch
    {
        "instanceId"      => Wrap(Instance.InstanceId),
        "workflowId"      => Wrap(Instance.WorkflowId),
        "workflowVersion" => Wrap(Instance.WorkflowVersion),
        "status"          => Wrap(Instance.Status.ToString()),
        "currentNodeId"   => Wrap(Instance.CurrentNodeId),
        "correlationId"   => Instance.CorrelationId != null
                                 ? Wrap(Instance.CorrelationId) : new NullContextValue(),
        "parentInstanceId"=> Instance.ParentInstanceId != null
                                 ? Wrap(Instance.ParentInstanceId) : new NullContextValue(),
        "createdAt"       => Wrap(Instance.CreatedAt.ToString("O")),
        "totalVisits"     => Wrap(TotalVisitCount),
        _ => null
    };

    private static JsonContextValue Wrap(object value)
        => new() { Value = JsonSerializer.SerializeToElement(value) };

    private static ContextValue? ResolveFromDict(
        IDictionary<string, ContextValue>? dict, string key)
    {
        if (dict == null) return null;
        var dot = key.IndexOf('.');
        if (dot < 0) return dict.GetValueOrDefault(key);
        var head = key[..dot];
        var tail = key[(dot + 1)..];
        if (!dict.TryGetValue(head, out var parent)) return null;
        return ResolveJsonPath(parent, "." + tail);
    }

    private static ContextValue? ResolveJsonPath(ContextValue? value, string subPath)
    {
        if (value is not JsonContextValue jv || string.IsNullOrEmpty(subPath)) return value;
        var node     = JsonNode.Parse(jv.Value.GetRawText());
        var selected = node?.SelectSingleNode("$" + subPath);
        if (selected == null) return null;
        return new JsonContextValue { Value = JsonSerializer.SerializeToElement(selected) };
    }

    private ContextValue SerialiseEngine()
        => new JsonContextValue { Value = JsonSerializer.SerializeToElement(new
            {
                instanceId    = Instance.InstanceId,
                workflowId    = Instance.WorkflowId,
                currentNodeId = Instance.CurrentNodeId,

21. SaveWithRetryAsync

csharp// Up to 3 attempts. On ConcurrencyException, reload fresh state and propagate our
// changes before retrying. Last attempt lets the exception propagate.
private async Task SaveWithRetryAsync(
    ExecutionInstance instance,
    string            claimantId,
    CancellationToken ct)
{
    const int maxRetries = 3;
    for (var i = 0; i < maxRetries; i++)
    {
        try
        {
            await _stateStore.SaveAsync(instance, ct);
            return;
        }
        catch (ConcurrencyException) when (i < maxRetries - 1)
        {
            var fresh = await _stateStore.LoadAsync(instance.InstanceId, ct);
            if (fresh == null) return;

            // Propagate our changes to the fresh snapshot
            fresh.Status          = instance.Status;
            fresh.CurrentNodeId   = instance.CurrentNodeId;
            fresh.UpdatedAt       = instance.UpdatedAt;
            fresh.CompletedAt     = instance.CompletedAt;
            fresh.Output          = instance.Output;
            fresh.Error           = instance.Error;
            fresh.PendingWait     = instance.PendingWait;
            fresh.Context         = instance.Context;
            fresh.NodeVisitCounts = instance.NodeVisitCounts;
            fresh.History         = instance.History;
            fresh.ClaimedBy       = claimantId;
            instance = fresh;
        }
    }
    await _stateStore.SaveAsync(instance, ct); // last attempt — let it throw
}

// ── AdvancePastCurrentNodeAsync ───────────────────────────────────────────────

/// <summary>
/// Advances CurrentNodeId past the current (wait/timer/human) node by evaluating
/// its edges with the given outcome. Called before ExecuteLoopAsync on resume so the
/// loop does not re-execute the suspension node — it starts at the next node instead.
/// On no matching edge: faults the instance.
/// </summary>
private async Task AdvancePastCurrentNodeAsync(
    ExecutionInstance instance, DecisionOutcome outcome, CancellationToken ct)
{
    var definition = await _workflowRegistry.GetAsync(
        instance.WorkflowId, instance.WorkflowVersion, ct);
    if (definition == null) return;
    if (!definition.Nodes.TryGetValue(instance.CurrentNodeId, out var node)) return;

    var ctx = BuildExecutionContext(instance, definition);

    foreach (var edge in node.Edges)
    {
        var outcomeMatches = edge.On == null
            || edge.On.Count == 0
            || edge.On.Contains(outcome);
        if (!outcomeMatches) continue;

        if (edge.When != null)
        {
            var whenMet = await _whenEvaluator.EvaluateAsync(edge.When, ctx, ct);
            if (!whenMet) continue;
        }

        instance.CurrentNodeId = edge.NextNodeId;
        instance.PendingWait   = null;
        instance.Status        = ExecStatus.Running;
        instance.UpdatedAt     = DateTimeOffset.UtcNow;
        return;
    }

    // No matching edge found
    await FaultInstanceAsync(instance, "NO_MATCHING_EDGE",
        $"No matching edge found for outcome '{outcome}' on node '{instance.CurrentNodeId}'",
        instance.CurrentNodeId);
}

/// <summary>
/// Writes signal payload to context:
///   _signal  — most recent payload (convenience for MinSignalCount=1)
///   _signals — append-only array of all received payloads
/// If the payload contains human-task response keys, promotes them to named entries:
///   _humanTaskResponse, _humanTaskResponses, _humanTaskConsensus
/// </summary>
private static void AppendSignalToContext(ExecutionInstance instance, JsonElement? signalPayload)
{
    if (!signalPayload.HasValue) return;

    instance.Context.Store("_signal",
        new JsonContextValue { Value = signalPayload.Value });

    var existing = instance.Context.Context.TryGetValue("_signals", out var sv)
        ? sv as JsonContextValue : null;
    var arr = existing?.Value.Deserialize<List<JsonElement>>() ?? new();
    arr.Add(signalPayload.Value);
    instance.Context.Store("_signals",
        new JsonContextValue { Value = JsonSerializer.SerializeToElement(arr) });

    // Promote human-task response keys when present
    if (signalPayload.Value.ValueKind == JsonValueKind.Object)
    {
        foreach (var contextKey in new[]
        {
            HumanTaskContextKeys.Response,
            HumanTaskContextKeys.Responses,
            HumanTaskContextKeys.Consensus
        })
        {
            if (signalPayload.Value.TryGetProperty(contextKey, out var prop)
                && prop.ValueKind != JsonValueKind.Undefined)
            {
                instance.Context.Store(contextKey,
                    prop.ValueKind == JsonValueKind.Null
                        ? new NullContextValue()
                        : new JsonContextValue { Value = prop });
            }
        }
    }
}

// ── ResumeFromSweepAsync ──────────────────────────────────────────────────────

/// <summary>
/// Called by ExecutionSweepService for crashed/timed-out/accumulated-signal instances.
/// Three distinct cases — see inline.
/// </summary>
internal async Task ResumeFromSweepAsync(string instanceId, CancellationToken ct)
{
    var instance = await _stateStore.LoadAsync(instanceId, ct);
    if (instance == null || IsTerminal(instance.Status)) return;

    // Case 1: Running with expired claim — server crashed; re-execute from saved state.
    // Node executors must be idempotent; use idempotencyKey on integration nodes.
    if (instance.Status == ExecStatus.Running)
    {
        await ExecuteLoopAsync(instance, ct);
        return;
    }

    if (instance.Status != ExecStatus.Suspended || instance.PendingWait == null)
        return;

    var now = DateTimeOffset.UtcNow;

    // Case 2: Accumulated enough signals (push path missed it)
    if (instance.PendingWait.ReceivedSignals.Count >= instance.PendingWait.MinSignalCount)
    {
        await AdvancePastCurrentNodeAsync(instance, DecisionOutcome.Match, ct);
        await _stateStore.SaveAsync(instance, ct);
        await ExecuteLoopAsync(instance, ct);
        return;
    }

    // Case 3: Wait timeout
    if (instance.PendingWait.TimeoutAt.HasValue && instance.PendingWait.TimeoutAt.Value <= now)
    {
        var definition = await _workflowRegistry.GetAsync(
            instance.WorkflowId, instance.WorkflowVersion, ct);

        string? nodeType = null;
        if (definition != null
            && definition.Nodes.TryGetValue(instance.CurrentNodeId, out var node))
            nodeType = node.Type;

        // Timer node: delay expiry is the intended path → Match (not an error)
        if (nodeType == "timer")
        {
            await AdvancePastCurrentNodeAsync(instance, DecisionOutcome.Match, ct);
            await _stateStore.SaveAsync(instance, ct);
            await ExecuteLoopAsync(instance, ct);
            return;
        }

        // Wait/human node timeout — jump to OnTimeoutNodeId or transition to TimedOut
        var onTimeoutNodeId = instance.PendingWait.OnTimeoutNodeId;
        instance.PendingWait = null;

        if (onTimeoutNodeId != null)
        {
            instance.Status        = ExecStatus.Running;
            instance.CurrentNodeId = onTimeoutNodeId;
            instance.UpdatedAt     = DateTimeOffset.UtcNow;
            await _stateStore.SaveAsync(instance, ct);
            await ExecuteLoopAsync(instance, ct);
        }
        else
        {
            instance.Status      = ExecStatus.TimedOut;
            instance.CompletedAt = DateTimeOffset.UtcNow;
            instance.UpdatedAt   = DateTimeOffset.UtcNow;
            instance.Error       = new ExecutionError
            {
                Code       = "WAIT_TIMEOUT",
                Message    = $"Wait node timed out at {now}",
                NodeId     = instance.CurrentNodeId,
                OccurredAt = now
            };
            await _stateStore.SaveAsync(instance, ct);
            EmitFinalEvent(instance);
        }
    }
}

// ── ResumeChildAsync ──────────────────────────────────────────────────────────

/// <summary>
/// Called by ChildSignalHandler after it has claimed the parent lock.
/// Enqueues the child signal, releases the child-signal lock, then runs the loop.
/// childSignalClaimantId is the claimant ID used by ChildSignalHandler so it can be released.
/// </summary>
internal async Task ResumeChildAsync(
    string            parentInstanceId,
    ChildSignal       signal,
    string            childSignalClaimantId,
    CancellationToken ct = default)
{
    // Enqueue the signal so loop/parallel executors can read it via GetChildrenAsync
    await _stateStore.EnqueueChildSignalAsync(parentInstanceId, signal, ct);

    // Release the lock acquired by ChildSignalHandler so we can re-acquire with our ClaimantId
    await _lockProvider.ReleaseAsync(parentInstanceId, childSignalClaimantId, ct);

    var parent = await _stateStore.LoadAsync(parentInstanceId, ct);
    if (parent == null || IsTerminal(parent.Status)) return;

    // Transition Suspended → Running so the loop continues
    if (parent.Status == ExecStatus.Suspended)
    {
        parent.Status    = ExecStatus.Running;
        parent.UpdatedAt = DateTimeOffset.UtcNow;
    }

    await ExecuteLoopAsync(parent, ct);
}
